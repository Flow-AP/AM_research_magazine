% ---------------------------------------------------------------- Makros
\newcommand{\rubrik}[1]{%
  \par\needspace{12\baselineskip}\bigskip
  {\color{amakzent}\rule{\linewidth}{1.2pt}}\par\smallskip
  {\sffamily\Large\bfseries\color{amblau}#1\par}%
  \addcontentsline{toc}{section}{#1}%
  \medskip
  \begin{multicols}{2}%
}
\newcommand{\rubrikende}{\end{multicols}}

\newcommand{\artikel}[6]{%
  \par\noindent\begin{minipage}[t]{\linewidth}%
  {\sffamily\bfseries\color{amblau}#1\par}\nopagebreak[4]\vspace{1pt}%
  {\footnotesize\sffamily\color{amgrau}#2%
    \if\relax\detokenize{#3}\relax\else\ \textbullet\ #3\fi\par}%
  \if\relax\detokenize{#5}\relax\else{\footnotesize\itshape\color{amgrau}#5\par}\fi
  \vspace{3pt}%
  {\small #6\par}\vspace{2pt}%
  {\footnotesize\sffamily\href{#4}{\color{amakzent}$\rightarrow$\ Zur Quelle}\par}%
  \vspace{6pt}{\color{amgrau!30}\rule{\linewidth}{0.4pt}}%
  \end{minipage}\par\vspace{8pt}%
}

\newcommand{\editorial}[1]{%
  \begin{center}
  \colorbox{amblau!7}{\parbox{0.94\linewidth}{\vspace{6pt}%
    {\sffamily\bfseries\color{amblau}Editorial\par}\smallskip
    \small #1\par\vspace{6pt}}}%
  \end{center}
}
